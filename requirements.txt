import streamlit as st
from google import genai

st.set_page_config(page_title="Kantor Virtual AI", page_icon="🏢")
st.title("🏢 Kantor Virtual AI")
st.markdown("Konsultasi bisnis praktis bersama: **Kang Roni** (Keuangan), **Bu Rina** (SDM), & **Dodit** (Kreatif).")

st.sidebar.title("⚙️ Pengaturan")
api_key = st.sidebar.text_input("Gemini API Key:", type="password")

persona = st.sidebar.selectbox("Pilih Ahli:", [
    "Kang Roni (Keuangan & Angka)", 
        "Bu Rina (SDM & Operasional)", 
            "Dodit (Pemasaran & Bisnis Kreatif)"
            ])

            prompts = {
                "Kang Roni (Keuangan & Angka)": "Anda adalah Kang Roni, pakar keuangan yang tegas, teliti, dan fokus pada efisiensi anggaran serta profit.",
                    "Bu Rina (SDM & Operasional)": "Anda adalah Bu Rina, manajer SDM yang ramah, tegas, terstruktur, dan disiplin.",
                        "Dodit (Pemasaran & Bisnis Kreatif)": "Anda adalah Dodit, ahli pemasaran yang santai, antusias, dan penuh ide kreatif out-of-the-box."
                        }

                        if "messages" not in st.session_state:
                            st.session_state.messages = []

                            for m in st.session_state.messages:
                                with st.chat_message(m["role"]):
                                        st.markdown(m["content"])

                                        if prompt := st.chat_input("Tulis pertanyaan bisnis Anda..."):
                                            if not api_key:
                                                    st.error("⚠️ Masukkan Google Gemini API Key di sidebar kiri!")
                                                        else:
                                                                st.session_state.messages.append({"role": "user", "content": prompt})
                                                                        with st.chat_message("user"):
                                                                                    st.markdown(prompt)
                                                                                            try:
                                                                                                        client = genai.Client(api_key=api_key)
                                                                                                                    contents = f"Karakter: {prompts[persona]}\n\n" + "\n".join([f"{x['role']}: {x['content']}" for x in st.session_state.messages])
                                                                                                                                with st.spinner("Berpikir..."):
                                                                                                                                                res = client.models.generate_content(model='gemini-2.5-flash', contents=contents)
                                                                                                                                                                reply = res.text
                                                                                                                                                                            with st.chat_message("assistant"):
                                                                                                                                                                                            st.markdown(reply)
                                                                                                                                                                                                        st.session_state.messages.append({"role": "assistant", "content": reply})
                                                                                                                                                                                                                except Exception as e:
                                                                                                                                                                                                                            st.error(f"Error: {e}")
